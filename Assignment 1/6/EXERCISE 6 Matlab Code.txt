% --- EXAMPLE A: INDEPENDENT ---
disp('=== EXAMPLE A: INDEPENDENT ===');
P_A = [0.28, 0.12; 
       0.42, 0.18];

px_A = sum(P_A, 2); % Row sums (marginal p(x))
py_A = sum(P_A, 1); % Column sums (marginal p(y))

hx_A   = compute_entropy(px_A);
hy_A   = compute_entropy(py_A);
hxy_A  = compute_entropy(P_A(:));

hx_given_y_A = hxy_A - hy_A;
hy_given_x_A = hxy_A - hx_A;
mi_A         = hx_A + hy_A - hxy_A;

fprintf('Marginal p(x) : [%.2f, %.2f]\n', px_A(1), px_A(2));
fprintf('Marginal p(y) : [%.2f, %.2f]\n', py_A(1), py_A(2));
fprintf('H(X)          : %.4f bits\n', hx_A);
fprintf('H(Y)          : %.4f bits\n', hy_A);
fprintf('H(X,Y)        : %.4f bits\n', hxy_A);
fprintf('H(X|Y)        : %.4f bits\n', hx_given_y_A);
fprintf('H(Y|X)        : %.4f bits\n', hy_given_x_A);
fprintf('I(X,Y)        : %.4f bits\n\n', mi_A);

% --- EXAMPLE B: DEPENDENT ---

% ======================================================
% LOCAL HELPER FUNCTION
% ======================================================
function h = compute_entropy(probs)
    probs = probs(probs > 0); % Filter out zeros to avoid log2(0)
    h = -sum(probs .* log2(probs));
end
